    //tHistorial opActual;

    while(operacionCompletada == 0)

        printf("\nOPERADOR>");

        }
        else if(strcmpi(op, "ENT") == 0)

        {
            codigoOp = CMD_REU;
        }
        else if(strcmpi(op, "ESP") == 0)
        {
            codigoOp = CMD_ESP;
        }
        else if(strcmpi(op, "VER") == 0)
        {
            codigoOp = CMD_VER;
        }
        else
            codigoOp = CMD_INVALIDO;

        switch(codigoOp)
        {
            case 1:
                operacionCompletada = 1;

                    operacionCompletada = 1;
                    ///opActual; --> se guarda el historial. y se encola
                }*/
                puts("ERROR DE ENTREGA.");
                break;
            case 3:
                operacionCompletada = 1;
                /*if(comandoReubicar())
                {
                    operacionCompletada = 1;
                    ///opActual; --> se guarda el historial. y se encola
                }*/
                puts("ERROR DE REUBICACION.");
                break;
            case 4:
                (*tiempo)++;
                operacionCompletada = 1;
                break;
            case 5:
                operacionCompletada = 1;
                /*if(comandoVer())
                {
                    operacionCompletada = 1;
                }*/
                puts("ERROR DE VISTA.");
                break;
            default:
                puts("OPERACION INVALIDA.");
                break;
        }


        //system("pause");
    }
}
